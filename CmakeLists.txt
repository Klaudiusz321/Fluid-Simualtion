# ============================================================
# SPH Fluid Simulation — CMake Build System
# ============================================================
# Requirements: CMake 3.20+, C++17 compiler, OpenGL 4.3+
#
# Build (MinGW on Windows):
#   cd build
#   cmake .. -G "MinGW Makefiles" -DCMAKE_BUILD_TYPE=Release
#   cmake --build .
#
# Build (MSVC on Windows):
#   cd build
#   cmake .. -DCMAKE_BUILD_TYPE=Release
#   cmake --build . --config Release
# ============================================================

cmake_minimum_required(VERSION 3.20)
project(SPHFluid VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# ============================================================
# Fetch GLFW via FetchContent
# ============================================================
include(FetchContent)

FetchContent_Declare(
    glfw
    GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG        3.4
)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(glfw)

# ============================================================
# GLAD (bundled — generated by glad2 for OpenGL 4.3 Core)
# ============================================================
add_library(glad STATIC
    ${CMAKE_SOURCE_DIR}/third_party/glad/src/gl.c
)
target_include_directories(glad PUBLIC
    ${CMAKE_SOURCE_DIR}/third_party/glad/include
)

# ============================================================
# Source Files
# ============================================================
set(CORE_SOURCES
    src/core/SPHSolver.cpp
    src/core/BoundaryHandler.cpp
)

set(PHYSICS_SOURCES
    src/physics/SurfaceTension.cpp
    src/physics/Viscosity.cpp
    src/physics/VorticityConfinement.cpp
)

set(RENDER_SOURCES
    src/render/Renderer.cpp
    src/render/FluidRenderer.cpp
)

set(ALL_SOURCES
    src/main.cpp
    ${CORE_SOURCES}
    ${PHYSICS_SOURCES}
    ${RENDER_SOURCES}
)

# ============================================================
# Executable
# ============================================================
add_executable(${PROJECT_NAME} ${ALL_SOURCES})

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/third_party/glad/include
)

# Find OpenGL & OpenMP
find_package(OpenGL REQUIRED)
find_package(OpenMP REQUIRED)

target_link_libraries(${PROJECT_NAME} PRIVATE
    glfw
    glad
    OpenGL::GL
    OpenMP::OpenMP_CXX
)

# ============================================================
# Copy shaders to build directory
# ============================================================
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_SOURCE_DIR}/shaders
    $<TARGET_FILE_DIR:${PROJECT_NAME}>/shaders
    COMMENT "Copying shaders to build directory"
)

# ============================================================
# Compiler-specific flags
# ============================================================
if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /W4 /O2 /fp:fast /MP)
else()
    target_compile_options(${PROJECT_NAME} PRIVATE 
        -Wall -Wextra -O2 -ffast-math
    )
    # MinGW needs these
    if(WIN32)
        target_link_libraries(${PROJECT_NAME} PRIVATE -static-libgcc -static-libstdc++)
    endif()
endif()

# ============================================================
# Installation
# ============================================================
install(TARGETS ${PROJECT_NAME} DESTINATION bin)
install(DIRECTORY shaders/ DESTINATION bin/shaders)

message(STATUS "")
message(STATUS "========================================")
message(STATUS "  SPH Fluid Simulation v${PROJECT_VERSION}")
message(STATUS "  C++ Standard: ${CMAKE_CXX_STANDARD}")
message(STATUS "  Build Type:   ${CMAKE_BUILD_TYPE}")
message(STATUS "========================================")
message(STATUS "")
